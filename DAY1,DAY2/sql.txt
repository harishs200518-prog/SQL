PART B – SQL JOINS
==================

-- INNER JOIN

-- 1. Display book names and member names for all borrowed books
SELECT b.book_name, m.member_name
FROM Books b
INNER JOIN Borrow br ON b.book_id = br.book_id
INNER JOIN Members m ON br.member_id = m.member_id;

-- 2. Display book name, member name, and borrow date
SELECT b.book_name, m.member_name, br.borrow_date
FROM Books b
INNER JOIN Borrow br ON b.book_id = br.book_id
INNER JOIN Members m ON br.member_id = m.member_id;

-- 3. Display book name, author, member name, and city
SELECT b.book_name, b.author, m.member_name, m.city
FROM Books b
INNER JOIN Borrow br ON b.book_id = br.book_id
INNER JOIN Members m ON br.member_id = m.member_id;

-- 4. Display all books borrowed by members from Chennai
SELECT b.book_name, m.member_name, m.city
FROM Books b
INNER JOIN Borrow br ON b.book_id = br.book_id
INNER JOIN Members m ON br.member_id = m.member_id
WHERE m.city = 'Chennai';

-- 5. Display all books borrowed by a specific member
SELECT b.book_name, m.member_name, br.borrow_date
FROM Books b
INNER JOIN Borrow br ON b.book_id = br.book_id
INNER JOIN Members m ON br.member_id = m.member_id
WHERE m.member_id = 201;

-- 6. Display member names who have borrowed Technology books
SELECT DISTINCT m.member_name
FROM Members m
INNER JOIN Borrow br ON m.member_id = br.member_id
INNER JOIN Books b ON br.book_id = b.book_id
WHERE b.category = 'Technology';

-- 7. Display book names and their corresponding borrowers
SELECT b.book_name, m.member_name
FROM Books b
INNER JOIN Borrow br ON b.book_id = br.book_id
INNER JOIN Members m ON br.member_id = m.member_id;

-- 8. Display borrow details sorted by borrow date
SELECT b.book_name, m.member_name, br.borrow_date
FROM Books b
INNER JOIN Borrow br ON b.book_id = br.book_id
INNER JOIN Members m ON br.member_id = m.member_id
ORDER BY br.borrow_date;


-- LEFT JOIN

-- 9. Display all books and the members who borrowed them
SELECT b.book_name, m.member_name
FROM Books b
LEFT JOIN Borrow br ON b.book_id = br.book_id
LEFT JOIN Members m ON br.member_id = m.member_id;

-- 10. Display all books including books that have never been borrowed
SELECT b.book_id, b.book_name, m.member_name
FROM Books b
LEFT JOIN Borrow br ON b.book_id = br.book_id
LEFT JOIN Members m ON br.member_id = m.member_id;

-- 11. Display all members and their borrowed books
SELECT m.member_name, b.book_name
FROM Members m
LEFT JOIN Borrow br ON m.member_id = br.member_id
LEFT JOIN Books b ON br.book_id = b.book_id;

-- 12. Display all members including members who have never borrowed a book
SELECT m.member_id, m.member_name, b.book_name
FROM Members m
LEFT JOIN Borrow br ON m.member_id = br.member_id
LEFT JOIN Books b ON br.book_id = b.book_id;

-- 13. Find books that have never been borrowed
SELECT b.book_id, b.book_name
FROM Books b
LEFT JOIN Borrow br ON b.book_id = br.book_id
WHERE br.book_id IS NULL;

-- 14. Find members who have never borrowed any book
SELECT m.member_id, m.member_name
FROM Members m
LEFT JOIN Borrow br ON m.member_id = br.member_id
WHERE br.member_id IS NULL;


-- RIGHT JOIN

-- 15. Display all borrow records and matching book names
SELECT br.borrow_id, br.book_id, b.book_name
FROM Books b
RIGHT JOIN Borrow br ON b.book_id = br.book_id;

-- 16. Display all borrow records with member names
SELECT br.borrow_id, br.member_id, m.member_name
FROM Members m
RIGHT JOIN Borrow br ON m.member_id = br.member_id;

-- 17. Display all members and their borrow information
SELECT br.borrow_id, m.member_id, m.member_name, br.borrow_date
FROM Borrow br
RIGHT JOIN Members m ON br.member_id = m.member_id;


-- CROSS JOIN

-- 18. Display every possible combination of books and members
SELECT b.book_name, m.member_name
FROM Books b
CROSS JOIN Members m;

-- 19. Count total combinations
SELECT COUNT(*) AS total_combinations
FROM Books b
CROSS JOIN Members m;

-- 20. Members and Technology books combinations
SELECT m.member_name, b.book_name
FROM Members m
CROSS JOIN Books b
WHERE b.category = 'Technology';


-- JOIN + GROUP BY

-- 21. Count books each member has borrowed
SELECT m.member_id, m.member_name, COUNT(br.borrow_id) AS books_borrowed
FROM Members m
LEFT JOIN Borrow br ON m.member_id = br.member_id
GROUP BY m.member_id, m.member_name;

-- 22. Number of members who borrowed each book
SELECT b.book_id, b.book_name, COUNT(DISTINCT br.member_id) AS member_count
FROM Books b
LEFT JOIN Borrow br ON b.book_id = br.book_id
GROUP BY b.book_id, b.book_name;

-- 23. Most borrowed book
SELECT b.book_id, b.book_name, COUNT(br.borrow_id) AS borrow_count
FROM Books b
JOIN Borrow br ON b.book_id = br.book_id
GROUP BY b.book_id, b.book_name
ORDER BY borrow_count DESC
LIMIT 1;

-- 24. Members who borrowed more than 2 books
SELECT m.member_id, m.member_name, COUNT(br.borrow_id) AS books_borrowed
FROM Members m
JOIN Borrow br ON m.member_id = br.member_id
GROUP BY m.member_id, m.member_name
HAVING COUNT(br.borrow_id) > 2;

-- 25. Categories and number of times borrowed
SELECT b.category, COUNT(br.borrow_id) AS borrow_count
FROM Books b
JOIN Borrow br ON b.book_id = br.book_id
GROUP BY b.category;

-- 26. Total borrowed books for each category
SELECT b.category, COUNT(*) AS total_borrowed_books
FROM Books b
JOIN Borrow br ON b.book_id = br.book_id
GROUP BY b.category;

-- D1 – CTE

-- 1. Average book price and books above average
WITH AveragePrice AS (
    SELECT AVG(price) AS avg_price
    FROM Books
)
SELECT b.*
FROM Books b
CROSS JOIN AveragePrice a
WHERE b.price > a.avg_price;

-- 2. Average price for each category
WITH CategoryAverage AS (
    SELECT category, AVG(price) AS average_price
    FROM Books
    GROUP BY category
)
SELECT *
FROM CategoryAverage;

-- 3. Books above category average
WITH CategoryAverage AS (
    SELECT category, AVG(price) AS average_price
    FROM Books
    GROUP BY category
)
SELECT b.*, ca.average_price
FROM Books b
JOIN CategoryAverage ca ON b.category = ca.category
WHERE b.price > ca.average_price;

-- 4. Total stock quantity for each category
WITH CategoryStock AS (
    SELECT category, SUM(stock_quantity) AS total_stock
    FROM Books
    GROUP BY category
)
SELECT *
FROM CategoryStock;

-- 5. Categories with stock greater than 20
WITH CategoryStock AS (
    SELECT category, SUM(stock_quantity) AS total_stock
    FROM Books
    GROUP BY category
)
SELECT *
FROM CategoryStock
WHERE total_stock > 20;

-- 6. Most expensive book in each category
WITH CategoryMaximum AS (
    SELECT category, MAX(price) AS max_price
    FROM Books
    GROUP BY category
)
SELECT b.*
FROM Books b
JOIN CategoryMaximum cm
    ON b.category = cm.category
   AND b.price = cm.max_price;

-- 7. Number of books written by each author
WITH AuthorBookCount AS (
    SELECT author, COUNT(*) AS book_count
    FROM Books
    GROUP BY author
)
SELECT *
FROM AuthorBookCount;

-- 8. Authors with more than one book
WITH AuthorBookCount AS (
    SELECT author, COUNT(*) AS book_count
    FROM Books
    GROUP BY author
)
SELECT *
FROM AuthorBookCount
WHERE book_count > 1;


-- D2 – WINDOW FUNCTIONS

-- 9. Rank by price descending
SELECT book_name, price,
       RANK() OVER (ORDER BY price DESC) AS price_rank
FROM Books;

-- 10. Rank by price ascending
SELECT book_name, price,
       RANK() OVER (ORDER BY price ASC) AS price_rank
FROM Books;

-- 11. ROW_NUMBER by price
SELECT book_name, price,
       ROW_NUMBER() OVER (ORDER BY price DESC) AS row_number
FROM Books;

-- 12. RANK within category
SELECT book_name, category, price,
       RANK() OVER (
           PARTITION BY category
           ORDER BY price DESC
       ) AS category_rank
FROM Books;

-- 13. DENSE_RANK within category
SELECT book_name, category, price,
       DENSE_RANK() OVER (
           PARTITION BY category
           ORDER BY price DESC
       ) AS category_rank
FROM Books;

-- 14. Category average price
SELECT book_name, category, price,
       AVG(price) OVER (PARTITION BY category) AS category_average
FROM Books;

-- 15. Highest price in category
SELECT book_name, category, price,
       MAX(price) OVER (PARTITION BY category) AS category_highest_price
FROM Books;

-- 16. Lowest price in category
SELECT book_name, category, price,
       MIN(price) OVER (PARTITION BY category) AS category_lowest_price
FROM Books;

-- 17. Difference from category average
SELECT book_name, category, price,
       AVG(price) OVER (PARTITION BY category) AS category_average,
       price - AVG(price) OVER (PARTITION BY category) AS price_difference
FROM Books;

-- 18. Cumulative stock
SELECT book_name, stock_quantity,
       SUM(stock_quantity) OVER (ORDER BY book_id) AS cumulative_stock
FROM Books;

-- 19. Cumulative stock per category
SELECT book_name, category, stock_quantity,
       SUM(stock_quantity) OVER (
           PARTITION BY category
           ORDER BY book_id
       ) AS cumulative_stock
FROM Books;

-- 20. Previous book price
SELECT book_name, price,
       LAG(price) OVER (ORDER BY book_id) AS previous_price
FROM Books;

-- 21. Next book price
SELECT book_name, price,
       LEAD(price) OVER (ORDER BY book_id) AS next_price
FROM Books;

-- 22. Difference from previous price
SELECT book_name, price,
       LAG(price) OVER (ORDER BY book_id) AS previous_price,
       price - LAG(price) OVER (ORDER BY book_id) AS price_difference
FROM Books;

-- 41. GetAllBooks
DROP FUNCTION IF EXISTS GetAllBooks();

CREATE OR REPLACE FUNCTION GetAllBooks()
RETURNS TABLE (
    book_id INT,
    book_name VARCHAR,
    author VARCHAR,
    price DECIMAL(10,2),
    category VARCHAR,
    stock_quantity INT
)
LANGUAGE plpgsql
AS $$
BEGIN
    RETURN QUERY
    SELECT
        b.book_id,
        b.book_name,
        b.author,
        b.price,
        b.category,
        b.stock_quantity
    FROM Books b;
END;
$$;

-- 1. Members who borrowed books above average price

FROM Members m
JOIN Borrow br ON m.member_id = br.member_id
JOIN Books b ON br.book_id = b.book_id

-- 2. Most expensive book in each category
WITH RankedBooks AS (
    SELECT b.*,
           RANK() OVER (
               PARTITION BY category
               ORDER BY price DESC
           ) AS rnk
    FROM Books b
)
SELECT book_name, category, price, author
FROM RankedBooks
WHERE rnk = 1;

-- 3. Categories above overall average price
SELECT category, AVG(price) AS average_price
FROM Books
GROUP BY category
HAVING AVG(price) > (SELECT AVG(price) FROM Books);

-- 4. Members who borrowed more than one book
SELECT m.member_id, m.member_name,
       COUNT(br.borrow_id) AS borrowed_count
FROM Members m
JOIN Borrow br ON m.member_id = br.member_id
GROUP BY m.member_id, m.member_name
HAVING COUNT(br.borrow_id) > 1;

SELECT b.*
FROM Books b
LEFT JOIN Borrow br ON b.book_id = br.book_id
WHERE br.book_id IS NULL
  AND b.price > 500;

    FROM Books b

ORDER BY b.category, b.price DESC;
